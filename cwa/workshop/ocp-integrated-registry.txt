# OCP 內建 Registry 補充路徑（2026-11-23 已於 OCP 4.20.39 驗證）
# 本檔取代主講義的 Registry 操作部分；其餘使用相同 app 與 render.py。
# Linux / Bash。以下各區段依序操作，不要整檔直接執行。
# 講師課前須準備 Project、對外 Registry Route、cwa-builder 推送身分，
# 以及 cwa-app 與 registry-pull 拉取認證。課中不要求學員操作叢集管理權限。
# 推送及拉取身分與 Token 請由講師配發，勿在教材中儲存 Token。

# 1. 填入講師提供的值；保留 TLS 驗證。
export PROJECT=cwa-user01
export REGISTRY=registry-route.example.invalid
export ARCH=amd64
export PULL_REGISTRY=image-registry.openshift-image-registry.svc:5000
export PUSH_REPO="$REGISTRY/$PROJECT/cwa-weather"
export DEPLOY_REPO="$PULL_REGISTRY/$PROJECT/cwa-weather"
export IMAGE_V1="$DEPLOY_REPO:1.0"
export IMAGE_V2="$DEPLOY_REPO:2.0"
export BASE_IMAGE=registry.access.redhat.com/ubi9/python-312:latest
oc project "$PROJECT"

# 2. 登入。互動輸入講師提供的短期推送 Token，Username 使用 cwa-builder。
mkdir -p .secrets
chmod 700 .secrets
podman login --authfile .secrets/registry-auth.json \
  --username cwa-builder "$REGISTRY"
chmod 600 .secrets/registry-auth.json
# 如講師已授予核發該 ServiceAccount Token 的權限，也可使用 stdin：
# oc create token cwa-builder --duration=1h | \
#   podman login --authfile .secrets/registry-auth.json \
#   --username cwa-builder --password-stdin "$REGISTRY"

# 3. 依 OCP 架構建置，再推送對外 Route。
podman build --platform "linux/$ARCH" --no-cache \
  --build-arg BASE_IMAGE="$BASE_IMAGE" --build-arg APP_VERSION=1.0 \
  -t "$PUSH_REPO:1.0" app
podman image inspect "$PUSH_REPO:1.0" \
  --format '{{.Os}}/{{.Architecture}}'
podman push --authfile .secrets/registry-auth.json "$PUSH_REPO:1.0"

# 4. Pod 改用叢集內 Registry 位址，避免繞出外網。
# render.py 預設建立 1Gi PVC；RWO + Recreate 更新時有短暫停機。
# registry-pull 的 auths 必須涵蓋 PULL_REGISTRY，不能只放對外 Route 的主機名。
oc get serviceaccount cwa-app -o yaml
python3 scripts/render.py --image "$DEPLOY_REPO:1.0"
oc apply --dry-run=server -f .generated/app.yaml
oc apply -f .generated/app.yaml
oc rollout status deployment/cwa-weather --timeout=180s
ROUTE_HOST=$(oc get route cwa-weather -o jsonpath='{.spec.host}')
curl -fsS "https://$ROUTE_HOST/observations"

# 5. 設定更新：先確認舊 Pod 的站名保持不變，再重啟讀取。
# 可先依 storage-lab.txt 做 volume 匯出、維護 Pod 匯入與 PVC 持久化驗證。
oc patch configmap cwa-weather-config --type=merge \
  -p '{"data":{"STATION_NAME":"Kaohsiung-DEMO"}}'
curl -fsS "https://$ROUTE_HOST/observations"
oc rollout restart deployment/cwa-weather
oc rollout status deployment/cwa-weather --timeout=180s
curl -fsS "https://$ROUTE_HOST/observations"

# 6. 2.0 釋出。仍推對外 Route，Deployment 使用叢集內位址。
podman build --platform "linux/$ARCH" --no-cache \
  --build-arg BASE_IMAGE="$BASE_IMAGE" --build-arg APP_VERSION=2.0 \
  -t "$PUSH_REPO:2.0" app
podman push --authfile .secrets/registry-auth.json "$PUSH_REPO:2.0"
oc set image deployment/cwa-weather api="$DEPLOY_REPO:2.0"
oc rollout status deployment/cwa-weather --timeout=180s
curl -fsS "https://$ROUTE_HOST/observations"

# 7. 緊接上一步回復 1.0；站名應保留 Kaohsiung-DEMO。
oc rollout history deployment/cwa-weather
oc rollout undo deployment/cwa-weather
oc rollout status deployment/cwa-weather --timeout=180s
curl -fsS "https://$ROUTE_HOST/observations"

# 注意：本次演練使用短期 ServiceAccount Token，重新演練前須重新核發。
# 對外 Route 與叢集內位址是同一個 Registry 的兩個入口。
# 若採 Quay / 外部 Registry，依主講義操作，不需使用此內建 Registry 分支。
