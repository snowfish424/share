# Podman 本機備課指令；Linux / Bash，工作目錄 workshop/
# 不需要 OCP、客戶網路、Registry 帳號或 API 金鑰。
# 首次取得基底映像需要連網；離線時由講師提供同架構映像。
# 請依區段操作，勿整檔執行。

# A. 環境與映像
bash scripts/preflight.sh --local
export BASE_IMAGE=registry.access.redhat.com/ubi9/python-312:latest
podman build --build-arg BASE_IMAGE="$BASE_IMAGE" \
  --build-arg APP_VERSION=1.0 -t localhost/cwa-weather:1.0 app
podman image inspect localhost/cwa-weather:1.0 \
  --format '{{.Os}}/{{.Architecture}}'
podman history localhost/cwa-weather:1.0

# B. 執行與設定，資料皆為模擬值
podman run -d --name cwa-local \
  -p 127.0.0.1:8080:8080 -e STATION_NAME=Taipei-DEMO \
  localhost/cwa-weather:1.0
curl -fsS http://127.0.0.1:8080/observations
podman logs cwa-local
podman exec cwa-local id
podman exec cwa-local cat /opt/app/data/startup.txt
podman stop cwa-local
podman rm cwa-local

# C. 非 root 與唯讀根目錄，12345 須可在主機 rootless UID 範圍內映射。
podman run -d --name cwa-local --user 12345:0 \
  --read-only --cap-drop ALL --security-opt no-new-privileges \
  --tmpfs /opt/app/data:rw,mode=0770 \
  --tmpfs /tmp \
  -p 127.0.0.1:8080:8080 -e STATION_NAME=Kaohsiung-DEMO \
  localhost/cwa-weather:1.0
curl -fsS http://127.0.0.1:8080/observations
podman exec cwa-local id
podman exec cwa-local cat /opt/app/data/startup.txt
podman stop cwa-local
podman rm cwa-local

# D. 讓 readiness 失敗，觀察它與 liveness 的差異
podman run -d --name cwa-local -p 127.0.0.1:8080:8080 \
  -e READY=false localhost/cwa-weather:1.0
curl -sS -o /dev/null -w '%{http_code}\n' http://127.0.0.1:8080/healthz
curl -sS -o /dev/null -w '%{http_code}\n' http://127.0.0.1:8080/readyz
# 預期依序為 200 與 503，程序仍然存活。
podman stop cwa-local
podman rm cwa-local

# E. 建立 2.0，保留 1.0
podman build --no-cache --build-arg BASE_IMAGE="$BASE_IMAGE" \
  --build-arg APP_VERSION=2.0 -t localhost/cwa-weather:2.0 app
podman run -d --name cwa-local -p 127.0.0.1:8080:8080 \
  localhost/cwa-weather:2.0
curl -fsS http://127.0.0.1:8080/observations
# 確認 version=2.0
podman stop cwa-local
podman rm cwa-local

# F. 本機映像封存與匯入，不刪除現有映像
podman tag localhost/cwa-weather:1.0 localhost/cwa-weather:archive-demo
podman save --format oci-archive -o cwa-weather-1.0.tar \
  localhost/cwa-weather:archive-demo
sha256sum cwa-weather-1.0.tar
# macOS 使用 shasum -a 256 cwa-weather-1.0.tar
podman load -i cwa-weather-1.0.tar
podman images
# 同一台主機重複匯入可確認格式；跨主機搬運需再核對校驗值與架構。
