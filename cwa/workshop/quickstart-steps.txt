# 學員精簡操作手順 / 2026-11-24
# 對應 06_學員精簡操作手順.pdf / .html。
# Linux / Bash；目前目錄必須是 workshop/。
# 講師已準備 oc 登入、個人 quickstart.env、Registry 推送 Token，
# 並已備妥 OCP 的 Project、應用身分、拉取認證與動態儲存。
# 以下各步驟逐段複製執行，每步確認成功再往下；不要整檔直接執行。
# 本流程建立全新 PVC，既有 volume 資料搬遷請見完整講義。

# 1. 確認環境
ls app/Containerfile scripts/render.py quickstart.env
source quickstart.env
oc whoami
oc project "$PROJECT"
printf 'Project=%s\nImage=%s\n' "$PROJECT" "$PUSH_IMAGE"
# 檢查點：確認目前 Project 與講師分配一致，映像為 quay.io/cwa/ocp/cwa-weather:1.0。

# 2-1. 檢視程式與 Containerfile
ls app/app.py app/requirements.txt app/Containerfile
cat app/Containerfile
# 檢查點：檔案均存在；Containerfile 含 requirements.txt 與 pip install。

# 2-2. 讀取 Python 套件清單
cat app/requirements.txt
# 檢查點：包含 requests==2.34.2，安裝流程見講義同頁。

# 2-3a. 用 Podman 打包映像
podman build --platform "linux/$ARCH" \
  --build-arg BASE_IMAGE="$BASE_IMAGE" \
  --build-arg APP_VERSION=1.0 \
  -f app/Containerfile \
  -t localhost/cwa-weather:1.0 app
# 檢查點：建置成功，輸出 localhost/cwa-weather:1.0。

# 2-3b. 檢查本機映像
podman images localhost/cwa-weather
podman image inspect localhost/cwa-weather:1.0 \
  --format '{{.Os}}/{{.Architecture}}'
# 檢查點：清單有 1.0；架構符合 ARCH。

# 2-3c. 驗證 Python 套件
podman run --rm --platform "linux/$ARCH" --read-only \
  --user 12345:0 --tmpfs /tmp:rw,size=16m \
  --entrypoint python3 localhost/cwa-weather:1.0 \
  -c 'import requests,sys,os; print(requests.__version__,sys.prefix,os.getuid())'
podman run --rm --platform "linux/$ARCH" --read-only \
  --entrypoint python3 localhost/cwa-weather:1.0 -m pip check
# 檢查點：顯示 2.34.2 /opt/venv 12345，pip check 成功。

# 2-4. 加上 Registry 名稱
podman tag localhost/cwa-weather:1.0 \
  quay.io/cwa/ocp/cwa-weather:1.0
podman image inspect localhost/cwa-weather:1.0 "$PUSH_IMAGE" \
  --format '{{.Id}}'
# 檢查點：兩個名稱列出相同的 Image ID；tag 後尚待步驟 3 推送。

# 3-1. 登入 Registry
mkdir -p .secrets
chmod 700 .secrets
podman login --authfile .secrets/registry-auth.json \
  --username "$REGISTRY_USER" quay.io
chmod 600 .secrets/registry-auth.json
# 檢查點：Password: 輸入講師提供的 Token；應看到 Login Succeeded。

# 3-2. 推送映像
podman push --authfile .secrets/registry-auth.json \
  quay.io/cwa/ocp/cwa-weather:1.0
# 檢查點：映像層與 manifest 推送成功，指令成功結束。

# 4. 部署 OCP
python3 scripts/render.py --image "$DEPLOY_IMAGE"
oc -n "$PROJECT" apply -f .generated/app.yaml
oc -n "$PROJECT" rollout status \
  deployment/cwa-weather --timeout=240s
oc -n "$PROJECT" get pods -l app=cwa-weather
oc -n "$PROJECT" get pvc cwa-weather-data
# 檢查點：rollout 成功；Pod 為 1/1 Running；PVC 為 Bound / 1Gi。

# 5. 開啟網址並驗收
ROUTE_HOST=$(oc -n "$PROJECT" get route cwa-weather \
  -o jsonpath='{.spec.host}')
APP_URL="https://$ROUTE_HOST/observations"
printf '%s\n' "$APP_URL"
curl -fsS "$APP_URL"
oc -n "$PROJECT" exec deployment/cwa-weather -- \
  python3 -c 'import requests; print(requests.__version__)'
# 檢查點：API 回應 version=1.0、sample_data=true、data_dir_writable=true；OCP 內 requests 為 2.34.2。

# 完成後保留應用與 PVC，交由講師檢視。
